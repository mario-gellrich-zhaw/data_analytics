import scraper_kit
from bs4 import BeautifulSoup
import re
import time

def is_apt_listing(listing):
    # Try to exclude ads for parking/commercial based on title, category, or similar fields if found
    title = listing.find('h2')
    if title and re.search(r'(park|garage|büro|office|lager|stellplatz|parking|shop|laden)', title.text, re.I):
        return False
    return True

def extract_listing_info(listing):
    data = {}
    # Try to pull out key fields, fallback to None if not found
    title = listing.find('h2')
    data['title'] = title.text.strip() if title else None
    attrs = listing.find_all('li')
    # Try guessing fields for room count, space, price
    for attr in attrs:
        text = attr.text.strip()
        if re.search(r'Zimmer', text, re.I):
            data['rooms'] = re.sub(r'[^\d\.]', '', text)
        elif re.search(r'm²', text):
            data['living_space_m2'] = re.sub(r'[^\d\.]', '', text)
        elif re.search(r'CHF', text):
            data['rent_gross_chf'] = re.sub(r'[^\d\.]', '', text)
    link = listing.find('a', href=True)
    data['url'] = 'https://www.immoscout24.ch' + link['href'] if link else None
    return data

def main():
    # ImmoScout24 Zurich rental search page 1
    url = 'https://www.immoscout24.ch/de/wohnung/mieten/ort-zuerich?pn=1'
    resp = scraper_kit.polite_get(url)
    print(f"Page status: {resp.status_code}")
    soup = BeautifulSoup(resp.text, 'html.parser')
    # Print the first page's structure to see key classes/tags
    print("First 1000 chars of page:", resp.text[:1000])
    listings = soup.find_all('article')
    print(f"Found {len(listings)} <article> tags; filtering...")
    apts = [l for l in listings if is_apt_listing(l)]
    print(f"Kept {len(apts)} after apartment filter")
    rows = []
    for apt in apts:
        row = extract_listing_info(apt)
        row['source'] = 'immoscout24.ch'
        rows.append(row)
    print(f"Saving {len(rows)} rows (sample: {rows[:1]})")
    scraper_kit.save_rows(rows)

if __name__ == '__main__':
    main()

import time

def filter_apt(listing):
    # Only want rental apartments, correct canton by zipcode
    if listing.get('offer_type') != 'RENT':
        return False
    if listing.get('object_category') != 'APARTMENT' and listing.get('object_category') != 'ATTIC_FLAT':
        return False
    zipc = listing.get('zipcode')
    return (zipc is not None and 8000 <= int(zipc) <= 8999)

def main():
    # First, find the total count of listings
    base_url = 'https://flatfox.ch/api/v1/public-listing/'

```python
import prep_kit
import pandas as pd
import numpy as np
import re
import scraper_kit

def extract_amenity(desc, keywords):
    if not isinstance(desc, str):
        return False
    desc_lower = desc.lower()
    return any(re.search(rf'\\b{kw}\\b', desc_lower) for kw in keywords)

def extract_features_from_description(df):
    # Example amenities to look for:
    amenity_keywords = {
        'balcony': ['balkon', 'balcony', 'terrasse', 'balustrade'],
        'garden': ['garten', 'garden', 'yard'],
        'elevator': ['lift', 'elevator', 'aufzug'],
        'dishwasher': ['dishwasher', 'geschirrspüler'],
        'parking': ['parkplatz', 'parking', 'garage'],
        'pets_allowed': ['haustiere erlaubt', 'pets allowed'],
        'washing_machine': ['washing machine', 'waschmaschine'],
    }
    for amenity, keywords in amenity_keywords.items():
        df[f'has_{amenity}'] = df['description'].apply(lambda x: extract_amenity(x, keywords) or extract_amenity(str(x), keywords))
    return df

def lookup_neighborhood(row):
    # Uses Swiss geo.admin API for municipality by coordinates
    if pd.isna(row['lat']) or pd.isna(row['lon']):
        return None
    try:
        params = {
            'geometry': f"{row['lon']},{row['lat']}",
            'geometryType': 'esriGeometryPoint',
            'sr': 4326,
            'tolerance': 0,
            'returnGeometry': 'false',
            'layers': 'all:ch.swisstopo.swissboundaries3d-gemeinde-flaeche.fill',
            'timeInstant': 2023
        }
        response = scraper_kit.polite_get(
            'https://api3.geo.admin.ch/rest/services/api/MapServer/identify', params=params)
        # Find the municipality name in the response structure
        j = response.json()
        features = j.get('results', [])
        if features and 'attributes' in features[0]:
            return features[0]['attributes'].get('gemeindename')
    except Exception as e:
        print(f"Lookup error: {e}")
    return None

def enrich_with_neighborhood(df):
    df['municipality'] = df.apply(lookup_neighborhood, axis=1)
    return df


df = prep_kit.load_data()
df = extract_features_from_description(df)
df = enrich_with_neighborhood(df)
# After extraction/drop of derived features, drop raw description for privacy
if 'description' in df:
    df = df.drop(columns=['description'])
prep_kit.save_data(df)
```

```python
import prep_kit
import pandas as pd
import numpy as np
import re
import scraper_kit

def extract_amenity(desc, keywords):
    if not isinstance(desc, str):
        return False
    desc_lower = desc.lower()
    return any(re.search(rf'\\b{kw}\\b', desc_lower) for kw in keywords)

def extract_features_from_description(df):
    amenity_keywords = {
        'balcony': ['balkon', 'balcony', 'terrasse', 'balustrade'],
        'garden': ['garten', 'garden', 'yard'],
        'elevator': ['lift', 'elevator', 'aufzug'],
        'dishwasher': ['dishwasher', 'geschirrspüler'],
        'parking': ['parkplatz', 'parking', 'garage'],
        'pets_allowed': ['haustiere erlaubt', 'pets allowed'],
        'washing_machine': ['washing machine', 'waschmaschine'],
    }
    for amenity, keywords in amenity_keywords.items():
        df[f'has_{amenity}'] = df['description'].apply(lambda x: extract_amenity(x, keywords) or extract_amenity(str(x), keywords))
    return df

def lookup_neighborhood(row):
    # Uses Swiss geo.admin API for municipality by coordinates
    if pd.isna(row['lat']) or pd.isna(row['lon']):
        return None
    try:
        params = {
            'geometry': f"{row['lon']},{row['lat']}",
            'geometryType': 'esriGeometryPoint',
            'sr': 4326,
            'tolerance': 0,
            'returnGeometry': 'false',
            'layers': 'all:ch.swisstopo.swissboundaries3d-gemeinde-flaeche.fill',
            'timeInstant': 2023
        }
        response = scraper_kit.polite_get(
            'https://api3.geo.admin.ch/rest/services/api/MapServer/identify', params=params)
        j = response.json()
        features = j.get('results', [])
        if features and 'attributes' in features[0]:
            return features[0]['attributes'].get('gemname')  # corrected from 'gemeindename' to 'gemname'
    except Exception as e:
        print(f"Lookup error: {e}")
    return None

def enrich_with_neighborhood(df):
    df['municipality'] = df.apply(lookup_neighborhood, axis=1)
    return df


df = prep_kit.load_data()
df = extract_features_from_description(df)
df = enrich_with_neighborhood(df)
# After extraction/drop of derived features, drop raw description for privacy
if 'description' in df:
    df = df.drop(columns=['description'])
prep_kit.save_data(df)
```
