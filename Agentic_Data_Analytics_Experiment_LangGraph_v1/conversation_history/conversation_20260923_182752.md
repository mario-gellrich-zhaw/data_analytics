import scraper_kit
from bs4 import BeautifulSoup
import re
import time

def parse_listing(item):
    title = item.select_one('.SearchList-Element__title').get_text(strip=True)
    url = 'https://www.immoscout24.ch' + item.select_one('a')['href']
    rent_gross = None
    rooms = None
    living_space_m2 = None
    street = None
    zip_code = None
    city = None
    year_built = None
    floor = None
    lat = None
    lon = None

    price_elem = item.select_one('.SearchList-Element__price')
    if price_elem:
        price_text = price_elem.get_text()
        m = re.search(r'(\d[\d\'\' ]+)\s*CHF', price_text)
        if m:
            rent_gross = int(m.group(1).replace("'", '').replace(' ', ''))
    
    detail_elems = item.select('.SearchList-Element__information .SearchList-Element__information-item')
    for elem in detail_elems:
        label = elem.get_text()
        if 'Zimmer' in label:
            try:
                rooms = float(label.split()[0])
            except Exception:
                pass
        elif 'm²' in label:
            try:
                living_space_m2 = int(float(label.split()[0]))
            except Exception:
                pass

    location_elem = item.select_one('.SearchList-Element__address')
    if location_elem:
        location_text = location_elem.get_text(strip=True)
        city_zip_match = re.search(r'(\d{4})\s+(.+)', location_text)
        if city_zip_match:
            zip_code = int(city_zip_match.group(1))
            city = city_zip_match.group(2)
        else:
            city = location_text

    # Save only Zurich canton (zip 8000-8999)
    if zip_code and (zip_code < 8000 or zip_code >= 9000):
        return None

    return {
        'source': 'immoscout24',
        'listing_id': url,
        'title': title,
        'rent_gross_chf': rent_gross,
        'rent_net_chf': None,
        'rent_charges_chf': None,
        'rooms': rooms,
        'living_space_m2': living_space_m2,
        'floor': floor,
        'year_built': year_built,
        'street': street,
        'zip': zip_code,
        'city': city,
        'lat': lat,
        'lon': lon,
        'url': url
    }

import time

def main():
    # Flatfox API returns JSON. We can filter by zipcode between 8000 and 8999 for Zurich
    # To get current listings, we need to page from the end (most recent listings at the end)
    url = 'https://flatfox.ch/api/v1/public-listing/'
    # First, get count
    count_page = scraper_kit.polite_get(url, params={'limit': 1, 'offset': 0})
    count_obj = json.loads(count_page.text)
    count = count_obj['count']
    print(f'Total listings found in Flatfox: {count}')
    
    batch = 100
    saved_total = 0
    dropped_price = 0
    dropped_category = 0
    dropped_zip = 0
    # Page backwards to fetch the most recent
    for offset in range(count - batch, -1, -batch):
        resp = scraper_kit.polite_get(url, params={'limit': batch, 'offset': offset})
        dat = json.loads(resp.text)
        results = dat.get('results', [])
        print(f'Pulled {len(results)} listings (offset {offset})')
        rows = []
        for item in results:
            # Only keep rental apartments in canton of Zurich (zip 8000-8999)

                dropped_zip += 1
                continue
            if item.get('object_category') != 'APARTMENT':

                continue

                continue

                dropped_price += 1
                continue

                'source': 'flatfox',
                'listing_id': item['pk'],

import scraper_kit
import json

def fetch_flatfox_listings(limit=100, offset=0):
    url = 'https://flatfox.ch/api/v1/public-listing/'
    params = {'limit': limit, 'offset': offset}
    page = scraper_kit.polite_get(url, params)
    if page.status_code != 200:
        print(f"Failed to fetch Flatfox: {page.status_code}")
        return None
    resp = page.json()
    print(f"flatfox.ch offset={offset}: {len(resp.get('results', []))} results (total={resp.get('count')})")
    return resp

def parse_listing(listing):
    try:
        # Only apartments for rent, Zurich zip
        if listing.get('offer_type') != 'RENT':
            return None
        if listing.get('object_category') not in ('APARTMENT', 'SHARED'):
            return None
        zip_code = listing.get('zipcode')
        if not (isinstance(zip_code, int) and 8000 <= zip_code < 9000):
            return None
        # Only if rent/rooms have value
        gross = listing.get('rent_gross')
        net = listing.get('rent_net')
        charges = listing.get('rent_charges')
        if gross is None and net is None and charges is None:
            return None
        rooms = listing.get('number_of_rooms')
        if not rooms:
            return None
        return {
            'source': 'flatfox',
            'listing_id': listing.get('pk'),
            'title': listing.get('title'),
            'rent_gross_chf': gross,
            'rent_net_chf': net,
            'rent_charges_chf': charges,
            'rooms': float(rooms.replace(',', '.')) if rooms else None,
            'living_space_m2': listing.get('surface_living'),
            'floor': listing.get('floor'),
            'year_built': listing.get('year_built'),
            'street': listing.get('street'),
            'zip': zip_code,
            'city': listing.get('city'),
            'lat': listing.get('latitude'),
            'lon': listing.get('longitude'),
            'url': f"https://flatfox.ch{listing.get('url')}" if listing.get('url') else None
        }
    except Exception as e:
        print(f"Failed to parse listing {listing.get('pk')}: {e}")
        return None

def main():
    # First, find total count to page from END
    resp = fetch_flatfox_listings(limit=1, offset=0)
    if not resp or 'count' not in resp:
        print("No count found on Flatfox API response.")
        return
    count = resp['count']
    print(f"Total listings in Flatfox: {count}")
    per_page = 100
    # Flatfox is oldest -> newest, so we start from the END for newest entries
    num_yielded = 0
    for offset in range(count - per_page, -1, -per_page):
        res = fetch_flatfox_listings(limit=per_page, offset=offset)
        if not res or 'results' not in res:
            continue
        records = []
        dropped = 0
        for l in res['results']:
            rec = parse_listing(l)
            if rec:
                records.append(rec)
            else:
                dropped += 1
        print(f"offset={offset}: Saved {len(records)}, Dropped {dropped} (non-apt/non-ZH/price-missing)")
        if records:
            scraper_kit.save_rows(records)
            num_yielded += len(records)
        if num_yielded >= 150:
            break

if __name__ == '__main__':
    main()
