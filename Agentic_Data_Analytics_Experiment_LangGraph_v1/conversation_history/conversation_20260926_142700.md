```python
import scraper_kit
import time
# Flatfox scraping (public API)
def is_valid_zip(zip_code):
    try:
        return 8000 <= int(zip_code) <= 8999
    except:
        return False

def process_flatfox_listing(item):
    # Only apartments for rent in canton Zurich
    if item.get('offer_type') != 'RENT':
        return None
    if item.get('object_category') not in ('APARTMENT', 'ATTIC_FLAT', 'STUDIO', 'DUPLEX'): 
        return None
    zipcode = item.get('zipcode')
    if not is_valid_zip(str(zipcode)):
        return None
    if item.get('rent_gross') is None and item.get('rent_net') is None and item.get('rent_charges') is None:
        return None  # must have at least one price
    attributes = ','.join(a['name'] for a in (item.get('attributes') or []) if 'name' in a)
    return {
        'source': 'flatfox.ch',
        'listing_id': item.get('pk'),
        'title': item.get('object_type'),
        'rent_gross_chf': item.get('rent_gross'),
        'rent_net_chf': item.get('rent_net'),
        'rent_charges_chf': item.get('rent_charges'),
        'rooms': item.get('number_of_rooms'),
        'living_space_m2': item.get('surface_living'),
        'floor': item.get('floor'),
        'year_built': item.get('year_built'),
        'street': item.get('street'),
        'zip': zipcode,
        'city': item.get('city'),
        'lat': item.get('latitude'),
        'lon': item.get('longitude'),
        'url': 'https://flatfox.ch' + str(item.get('url')) if item.get('url') else None,
        'object_type': item.get('object_type'),
        'description': item.get('description'),
        'attributes': attributes,
        'is_furnished': item.get('is_furnished'),
        'year_renovated': item.get('year_renovated'),
        'moving_date': item.get('moving_date'),
    }

# Initial call: get total count
data_url = 'https://flatfox.ch/api/v1/public-listing/?limit=1&offset=0'
data_page = scraper_kit.polite_get(data_url)
data = data_page.json()
total = data['count']
print(f"Total listings on Flatfox: {total}")

# Page backwards from the end for freshest Zurich apartments
BATCH_SIZE = 100
max_rows = 150  # script budget
collected = 0
for offset in range(total - BATCH_SIZE, -1, -BATCH_SIZE):
    url = f'https://flatfox.ch/api/v1/public-listing/?limit={BATCH_SIZE}&offset={offset}'
    page = scraper_kit.polite_get(url)
    items = page.json()['results']
    to_save = []
    dropped = 0
    for item in items:
        parsed = process_flatfox_listing(item)
        if parsed:
            to_save.append(parsed)
        else:
            dropped += 1
        if len(to_save) + collected >= max_rows:
            break
    print(f"Page offset {offset}: {len(to_save)} listings saved, {dropped} dropped.")
    if to_save:
        scraper_kit.save_rows(to_save)
        collected += len(to_save)
    if collected >= max_rows:
        break
    time.sleep(2) # polite pause
```

```python
import prep_kit
import pandas as pd
import numpy as np

def clean_floor(val):
    if pd.isna(val):
        return np.nan
    try:
        f = float(val)
        # Accept Swiss-style basements (-1), normal floors 0-40
        if -1 <= f <= 40:
            return f
        else:
            return np.nan
    except Exception:
        return np.nan

def clean_whitespace(val):
    if isinstance(val, str):
        return val.strip()
    return val

# Load data
df = prep_kit.load_data()
rows_before = len(df)

# Drop listings with missing gross rent
missing_gross_rent = df['rent_gross_chf'].isna().sum()
df = df[~df['rent_gross_chf'].isna()]
rows_after_gross_rent = len(df)
print(f"Dropped {missing_gross_rent} rows missing gross rent.")

# Clean floor
invalid_floor = df.loc[~df['floor'].isna() & ((df['floor'] < -1) | (df['floor'] > 40)), 'listing_id'].tolist()
df['floor'] = df['floor'].apply(clean_floor)
print(f"Set floor to NaN for {len(invalid_floor)} rows with implausible values: {invalid_floor}.")

# Clean up whitespace in string columns
to_strip = ['title', 'street', 'city', 'object_type', 'description', 'attributes']
for col in to_strip:
    df[col] = df[col].apply(clean_whitespace)

# Ensure correct types
df['listing_id'] = df['listing_id'].astype(int)

# Save cleaned data
prep_kit.save_data(df)
print(f"Rows before cleaning: {rows_before}, after cleaning: {len(df)}.")
```
