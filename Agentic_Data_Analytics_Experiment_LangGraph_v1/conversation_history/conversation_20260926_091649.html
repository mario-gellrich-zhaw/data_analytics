import prep_kit
import pandas as pd
import numpy as np
import re
import scraper_kit

def enrich(df):
    # Feature extraction from attributes & description
    amenity_keys = ["balconygarden", "minergie", "lift", "accessiblewithwheelchair", "parkingspace", "parquetflooring", "petsallowed", "garage"]
    for key in amenity_keys:
        df[f"has_{key}"] = df["attributes"].fillna('').apply(lambda x: key in x.split(","))
    # Description features
    desc_keys = ["renoviert", "erstbezug", "ruhe", "hell", "zentral", "einbauküche", "altbau", "ausblick", "luxuriös", "ruhig"]
    for key in desc_keys:
        df[f"desc_{key}"] = df["description"].str.lower().fillna("").str.contains(key)
    
    # Lookup: is any public transport stop within 1km, via geo.admin.ch layer ch.bav.haltestellen-oev
    def has_public_transport(lat, lon):
        if pd.isnull(lat) or pd.isnull(lon):
            return False
        try:
            w, h = 500, 500
            tol = 1000
            minlat, minlon = lat-0.01, lon-0.01
            maxlat, maxlon = lat+0.01, lon+0.01
            params = {
                "geometry": f"{lon},{lat}",
                "geometryType": "esriGeometryPoint",
                "sr": 4326,
                "tolerance": tol,
                "returnGeometry": "false",
                "layers": "all:ch.bav.haltestellen-oev",
                "mapExtent": f"{minlon},{minlat},{maxlon},{maxlat}",
                "imageDisplay": f"{w},{h},96"
            }
            page = scraper_kit.polite_get("https://api3.geo.admin.ch/rest/services/api/MapServer/identify", params=params)
            stops = page.json().get("results", [])
            return len(stops) > 0
        except Exception:
            return False
    df["has_public_transport_within_1km"] = df.apply(lambda row: has_public_transport(row["lat"], row["lon"]), axis=1)
    # Drop raw description now that features are extracted
    df = df.drop(columns=["description"])
    return df

df = prep_kit.load_data()
df = enrich(df)
prep_kit.save_data(df)
