import time

def extract_integer(text):
    if text:
        match = re.search(r'\d+', text.replace("'", ''))
        if match:
            return int(match.group(0))
    return None

def extract_float(text):
    if text:
        match = re.search(r'[\d.]+', text.replace(',', '.'))
        if match:
            return float(match.group(0))
    return None
