import scraper_kit
import re
import time
from bs4 import BeautifulSoup

def extract_integer(text):
    if text:
        match = re.search(r'\d+', text.replace("'", ''))
        if match:
            return int(match.group(0))
    return None

def extract_float(text):
    if text:
        match = re.search(r'[\d.]+', text.replace(',', '.'))
        if match:
            return float(match.group(0))
    return None

import prep_kit
import pandas as pd

df = prep_kit.load_data()

# Feature: rent per square meter
with np.errstate(divide='ignore', invalid='ignore'):
    df['rent_per_m2'] = df['rent_gross_chf'] / df['living_space_m2']

# Feature extraction helpers
def extract_feature(src, keywords):
    if isinstance(src, str):
        src = src.lower()
        return int(any(keyword in src for keyword in keywords))
    return 0

def extract_from_both(row, keywords):
    # attributes is a comma-separated string
    attrs = row['attributes'] if pd.notnull(row['attributes']) else ''
    desc = row['description'] if pd.notnull(row['description']) else ''
    return max(extract_feature(attrs, keywords), extract_feature(desc, keywords))

# Extract amenity features
balcony_keywords = ['balkon', 'balcony', 'terrasse', 'terrace', 'loggia']
elevator_keywords = ['lift', 'elevator', 'aufzug']
parking_keywords = ['garage', 'parkplatz', 'car park', 'parking']
wheelchair_keywords = ['rollstuhl', 'wheelchair', 'behindertengerecht']
garden_keywords = ['garten', 'garden']

for name, keys in [
    ('has_balcony', balcony_keywords),
    ('has_elevator', elevator_keywords),
    ('has_parking', parking_keywords),
    ('is_wheelchair_accessible', wheelchair_keywords),
    ('has_garden', garden_keywords),
]:
    df[name] = df.apply(lambda row: extract_from_both(row, keys), axis=1)

def lookup_municipality(lat, lon):
    if pd.isnull(lat) or pd.isnull(lon):
        return None
    try:
        resp = polite_get(
            "https://api3.geo.admin.ch/rest/services/api/MapServer/identify",
            params={
                "geometry": f"{lon},{lat}",
                "geometryType": "esriGeometryPoint",
                "sr": 4326,
                "tolerance": 0,
                "returnGeometry": "false",
                "layers": "all:ch.swisstopo.swissboundaries3d-gemeinde-flaeche.fill",
                "timeInstant": "2023"
            }
        )
        data = resp.json()
        # The response structure: from a first run we'll print what keys are present

    return None

# For initial run: only lookup for the first 5 rows to print structure
