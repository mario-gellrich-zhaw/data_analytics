```python
import prep_kit
import pandas as pd
import numpy as np
import re
from datetime import datetime
import scraper_kit

def extract_feature_presence(text, keywords):
    if pd.isna(text):
        return {key: False for key in keywords}
    text_lower = text.lower()
    return {key: (key in text_lower) for key in keywords}

def min_distance_to_station(lat, lon):
    # Look up closest public transport station (Zurich VBZ S-Bahn/Tram/Bus) within 1km radius using geo.admin.ch service
    # Using Zurich transport layer: ch.bav.haltestellen-oev
    # Returns minimal distance in meters
    try:
        params = {
            "geometry": f"{lon},{lat}",
            "geometryType": "esriGeometryPoint",
            "sr": 4326,
            "tolerance": 1000,  # Large enough radius for city walk
            "returnGeometry": "false",
            "layers": "all:ch.bav.haltestellen-oev"
        }
        page = scraper_kit.polite_get("https://api3.geo.admin.ch/rest/services/api/MapServer/identify", params=params)
        stops = page.json().get("results", [])
        if not stops:
            return np.nan
        # Calculate distance to each found stop
        dists = []
        for stop in stops:
            if "geometry" in stop:
                pt = stop["geometry"]
                s_lat, s_lon = pt["y"], pt["x"]
                dist = ((lat-s_lat)**2 + (lon-s_lon)**2)**0.5 * 111000  # crude approx in meters
                dists.append(dist)
        return min(dists) if dists else np.nan
    except Exception as e:
        return np.nan

def enrich(df):
    # Feature extraction from attributes & description
    amenity_keys = ["balconygarden", "minergie", "lift", "accessiblewithwheelchair", "parkingspace", "parquetflooring", "petsallowed", "garage"]
    for key in amenity_keys:
        df[f"has_{key}"] = df["attributes"].fillna('').apply(lambda x: key in x.split(","))
    # Description features
    desc_keys = ["renoviert", "erstbezug", "ruhe", "hell", "zentral", "einbauküche", "altbau", "ausblick", "luxuriös", "ruhig"]
    for key in desc_keys:
        df[f"desc_{key}"] = df["description"].str.lower().fillna("").str.contains(key)
    # Lookup minimum distance to a public transport stop for each apartment
    df["min_dist_public_transport_m"] = df.apply(lambda row: min_distance_to_station(row["lat"], row["lon"]) if pd.notnull(row["lat"]) and pd.notnull(row["lon"]) else np.nan, axis=1)
    # Drop raw description now that features are extracted
    df = df.drop(columns=["description"])
    return df

df = prep_kit.load_data()
df = enrich(df)
prep_kit.save_data(df)
```

```python
def min_distance_to_station(lat, lon):
    # Look up closest public transport station (Zurich VBZ S-Bahn/Tram/Bus) within 1km radius using geo.admin.ch service
    # Using Zurich transport layer: ch.bav.haltestellen-oev
    # Returns minimal distance in meters

        tol = 1000

            "tolerance": tol,
            "returnGeometry": "false",

            "mapExtent": f"{minlon},{minlat},{maxlon},{maxlat}",
            "imageDisplay": f"{w},{h},96"
        }
        page = scraper_kit.polite_get("https://api3.geo.admin.ch/rest/services/api/MapServer/identify", params=params)
        stops = page.json().get("results", [])
        if not stops:

        # Calculate distance to each found stop
        dists = []
        for stop in stops:

                dist = ((lat-s_lat)**2 + (lon-s_lon)**2)**0.5 * 111000  # crude approx in meters
```
